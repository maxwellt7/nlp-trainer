```js
import db from '../db/index.js';
import { v4 as uuidv4 } from 'uuid';
import { safeJsonParse } from './safe-json.js';

function rowToGoal(row) {
  if (!row) return null;
  return {
    id: row.id,
    title: row.title,
    description: row.description || '',
    category: row.category || 'Health',
    targetDate: row.target_date || '',
    milestones: safeJsonParse(row.milestones, []),
    rewards: safeJsonParse(row.rewards, []),
    completed: !!row.completed,
    createdAt: row.created_at,
    updatedAt: row.updated_at,
  };
}

function cleanRewards(rewards) {
  if (!Array.isArray(rewards)) return [];
  return rewards
    .filter(r => (r && ((r.name && r.name.trim()) || r.image)))
    .map(r => ({ type: r.type === 'experience' ? 'experience' : 'material', name: (r.name || '').trim(), image: r.image || '' }));
}

function cleanMilestones(ms) {
  if (!Array.isArray(ms)) return [];
  return ms.filter(m => m && typeof m.text === 'string' && m.text.trim())
    .map(m => ({ text: m.text.trim(), done: !!m.done }));
}

export function listGoals(userId) {
  const rows = db.prepare('SELECT * FROM goals WHERE user_id = ? ORDER BY target_date ASC').all(userId);
  return rows.map(rowToGoal);
}

export function getGoal(userId, id) {
  return rowToGoal(db.prepare('SELECT * FROM goals WHERE id = ? AND user_id = ?').get(id, userId));
}

export function createGoal(userId, draft = {}) {
  const id = `goal-${uuidv4()}`;
  db.prepare(`INSERT INTO goals (id, user_id, title, description, category, target_date, milestones, rewards, completed)
              VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?)`).run(
    id, userId,
    (draft.title || '').trim(),
    draft.description || '',
    draft.category || 'Health',
    draft.targetDate || '',
    JSON.stringify(cleanMilestones(draft.milestones)),
    JSON.stringify(cleanRewards(draft.rewards)),
    draft.completed ? 1 : 0,
  );
  return getGoal(userId, id);
}

export function updateGoal(userId, id, patch = {}) {
  const existing = getGoal(userId, id);
  if (!existing) return null;
  const next = {
    title: patch.title !== undefined ? patch.title.trim() : existing.title,
    description: patch.description !== undefined ? patch.description : existing.description,
    category: patch.category !== undefined ? patch.category : existing.category,
    target_date: patch.targetDate !== undefined ? patch.targetDate : existing.targetDate,
    milestones: patch.milestones !== undefined ? cleanMilestones(patch.milestones) : existing.milestones,
    rewards: patch.rewards !== undefined ? cleanRewards(patch.rewards) : existing.rewards,
    completed: patch.completed !== undefined ? (patch.completed ? 1 : 0) : (existing.completed ? 1 : 0),
  };
  db.prepare(`UPDATE goals SET title=?, description=?, category=?, target_date=?, milestones=?, rewards=?, completed=?, updated_at=datetime('now')
              WHERE id=? AND user_id=?`).run(
    next.title, next.description, next.category, next.target_date,
    JSON.stringify(next.milestones), JSON.stringify(next.rewards), next.completed, id, userId,
  );
  return getGoal(userId, id);
}

export function deleteGoal(userId, id) {
  const existing = getGoal(userId, id);
  if (!existing) return false;
  db.prepare('DELETE FROM goals WHERE id = ? AND user_id = ?').run(id, userId);
  return true;
}
```

- [ ] **Step 1: Implement** `server/routes/goals.js` (mirror `server/routes/profile.js` error shape):
```js
import { Router } from 'express';
import { listGoals, createGoal, updateGoal, deleteGoal } from '../services/goals.js';

const router = Router();

router.get('/', (req, res) => {
  try { res.json({ goals: listGoals(req.userId) }); }
  catch (e) { console.error('Error listing goals:', e.message); res.status(500).json({ error: 'Failed to list goals' }); }
});

router.post('/', (req, res) => {
  try {
    if (!req.body || !req.body.title || !req.body.title.trim()) return res.status(400).json({ error: 'Title is required' });
    res.status(201).json({ goal: createGoal(req.userId, req.body) });
  } catch (e) { console.error('Error creating goal:', e.message); res.status(500).json({ error: 'Failed to create goal' }); }
});

router.put('/:id', (req, res) => {
  try {
    const goal = updateGoal(req.userId, req.params.id, req.body || {});
    if (!goal) return res.status(404).json({ error: 'Goal not found' });
    res.json({ goal });
  } catch (e) { console.error('Error updating goal:', e.message); res.status(500).json({ error: 'Failed to update goal' }); }
});

router.delete('/:id', (req, res) => {
  try {
    if (!deleteGoal(req.userId, req.params.id)) return res.status(404).json({ error: 'Goal not found' });
    res.json({ success: true });
  } catch (e) { console.error('Error deleting goal:', e.message); res.status(500).json({ error: 'Failed to delete goal' }); }
});

export default router;
```

export interface GoalReward { type: 'material' | 'experience'; name: string; image: string; }
export interface GoalMilestone { text: string; done: boolean; }
export interface Goal {

import type { Goal } from '../../services/api';

export const GOAL_CATS = [
  { name: 'Health', color: '#A9C6B3' },
  { name: 'Wealth', color: '#7FB4D9' },
  { name: 'Family', color: '#C7A9B4' },
  { name: 'Craft', color: '#9DB4CC' },
  { name: 'Adventure', color: '#B3D6EC' },
  { name: 'Spirit', color: '#A5A3C9' },
];

export function catColor(name: string): string {
  return (GOAL_CATS.find(c => c.name === name) || GOAL_CATS[0]).color;
}

export function currentAge(birthYear: number, thisYear: number): number {
  return Math.max(0, thisYear - birthYear);
}

export interface GoalCard {
  id: string; hasImage: boolean; image: string; catColor: string;
  metaLine: string; title: string; thumbs: string[]; rewardLine: string;
  showProgress: boolean; progressPct: string; achieved: boolean;
}

export function goalCard(g: Goal, birthYear: number, thisYear: number): GoalCard {
  const img = (g.rewards.find(r => r.image) || { image: '' }).image;
  const year = parseInt((g.targetDate || '').slice(0, 4)) || thisYear;
  const age = year - birthYear;
  const total = g.milestones.length;
  const done = g.milestones.filter(m => m.done).length;
  const prog = g.completed ? 100 : (total ? Math.round((done / total) * 100) : 0);
  const thumbs = g.rewards.filter(r => r.image).slice(0, 3).map(r => r.image);
  const named = g.rewards.filter(r => r.name && r.name.trim());
  const rewardLine = !g.rewards.length ? ''
    : (g.rewards.length === 1 ? (named[0] ? named[0].name : 'A reward waiting') : `${g.rewards.length} rewards waiting`);
  return {
    id: g.id, hasImage: !!img, image: img || '', catColor: catColor(g.category),
    metaLine: `${year} · age ${age} · ${g.category}`, title: g.title,
    thumbs, rewardLine, showProgress: prog > 0, progressPct: `${prog}%`, achieved: g.completed,
  };
}

export interface Decade {
  d0: number; label: string; years: string; isNow: boolean; hasGoals: boolean;
  goals: GoalCard[]; addLabel: string; prefillYear: number;
}

export function buildDecades(goals: Goal[], birthYear: number, thisYear: number): Decade[] {
  const curAge = currentAge(birthYear, thisYear);
  const startDec = Math.min(80, Math.max(0, Math.floor(curAge / 10) * 10));
  const out: Decade[] = [];
  for (let d0 = startDec; d0 <= 80; d0 += 10) {
    const gs = goals.filter(g => {
      const y = parseInt((g.targetDate || '').slice(0, 4)) || thisYear;
      const a = Math.min(89, Math.max(curAge, y - birthYear));
      return Math.max(startDec, Math.floor(a / 10) * 10) === d0;
    }).sort((a, b) => (a.targetDate || '').localeCompare(b.targetDate || ''));
    const isNow = curAge >= d0 && curAge < d0 + 10;
    out.push({
      d0, label: `${d0}s`, years: `${birthYear + d0} – ${birthYear + d0 + 9}`,
      isNow, hasGoals: gs.length > 0,
      goals: gs.map(g => goalCard(g, birthYear, thisYear)),
      addLabel: gs.length ? '+ add another' : `Add a goal for your ${d0}s`,
      prefillYear: isNow ? thisYear + 1 : birthYear + d0 + 2,
    });
  }
  return out;
}

export function draftValid(d: { title: string; targetDate: string }): boolean {
  return d.title.trim().length > 0 && !!d.targetDate;
}

export function ageAtLine(birthYear: number, targetDate: string): string {
  if (!targetDate) return '';
  const year = parseInt(targetDate.slice(0, 4));
  if (!year) return '';
  return `You'll be ${year - birthYear} when this comes true.`;
}
